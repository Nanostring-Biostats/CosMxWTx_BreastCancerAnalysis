4  Major tumor lobe differences

In this section we look to identify the major transcriptional differences between tumor cells in the lower dense lobe and tumor cells in the upper interdigitated lobe of the tumor.

4.1 Identify tumor lobes

We first manually define the two main regions of the tumor - the bottom dense, localized lobe as well as the upper interdigitated lobe.

Manually draw a polygon around the lower localized tumor lobe.

Show the code 
# if there's no seurat object, read it in
if (!("seu.obj" %in% ls())){
  seu.obj <- readRDS(results_list$seurat_file_postCellTyping)
}

if(is.null(results_list$lower_lobe_cells)){
  
  ## Select outermost ring of cells to consider
  
  # Define a polygon selection function
  euclidean <- function(a, b) {
    sqrt(sum((a - b)^2))
  }
  
  select_polygon <- function(distmin = 0.01, max_points = 40){
    message("Please select points to define a polygon.  Click to the first point to stop.  First point marker is largest.")
    lims <- par("usr")
    margins <- par("plt")
    calcX <- function(x, limitss = lims, mars = margins) limitss[1] + ((x-mars[1])/(1-(1-mars[2])-mars[1]))*(limitss[2]-limitss[1])
    calcY <- function(y, limitss = lims, mars = margins) limitss[3] + ((y-mars[3])/(1-(1-mars[4])-mars[3]))*(limitss[4]-limitss[3])
    calcXY <- function(z) c(calcX(z[1]), calcY(z[2]))
    
    # determine diagonal distance in the plotting coordinate frame
    diagonalDist <- euclidean(lims[c(1,3)], lims[c(2,4)])
    
    # select first and second point and third point
    pg <- matrix(c(0,0), nrow = 1)
    pg[1, ] <- calcXY(unlist(grid::grid.locator(unit = "npc")))
    points(pg[1,1], pg[1,2], cex = 1, pch = 9, col = "magenta")
    pg <- rbind(pg, calcXY(unlist(grid::grid.locator(unit = "npc"))))
    points(pg[nrow(pg),1], pg[nrow(pg),2], cex = 0.5, pch = 9, col = "magenta")
    pg <- rbind(pg, calcXY(unlist(grid::grid.locator(unit = "npc"))))
    points(pg[nrow(pg),1], pg[nrow(pg),2], cex = 0.5, pch = 9, col = "magenta")
    
    #select additional points until a point is within the desired distance of the original point (or hit max_points limit)
    while(euclidean(pg[nrow(pg),], pg[1,]) > distmin*diagonalDist){
      pg <- rbind(pg, calcXY(unlist(grid::grid.locator(unit = "npc"))))
      points(pg[nrow(pg),1], pg[nrow(pg),2], cex = 0.5, pch = 9, col = "magenta")
      if(nrow(pg) > max_points) {
        message("You reached the max allowed points.")
        break
      }
    }
    if(nrow(pg) > 3) pg <- pg[1:(nrow(pg)-1),]
    polygon(pg[,1],pg[,2],col="#00000007",border="red")
    return(pg)
  }
  
  # Plot cells in bottom of tumor
  plot(seu.obj@meta.data$x_slide_mm, 
       seu.obj@meta.data$y_slide_mm, 
       cex = 0.01, ylim = c(-0.5,6), 
       asp=1, col = rgb(0,0,0,0.1))
  
  
  # Select outer tumor edge polygon manually
  pg_outer <- select_polygon()
  
  # which points are within the polygon
  idx_selected <- mgcv::in.out(pg_outer, as.matrix(seu.obj@meta.data[,c("x_slide_mm","y_slide_mm")] ))
  outer_cells <- seu.obj@meta.data$cell_id[idx_selected]
  
  # Add to results list
  results_list$lower_lobe_cells <- outer_cells
  saveRDS(results_list, results_list_file)
}

4.3 Contrast tumor lobes - differential expression

We use differential expression to contrast tumor cells in the two lobes, using only genes that are more likely to be expressed in tumor cells than surrounding cells (ie are unlikely to be present simply due to tumor cells overlapping with non-tumor cells at their borders).

Show the code 
if (!("seu.obj" %in% ls())){
  seu.obj <- readRDS(results_list$seurat_file_postCellTyping)
}

# Isolate meta for plotting
meta <- seu.obj@meta.data

# Add lobe assignments
meta$lobe <- "none"
meta[results_list$upper_lobe_cells, "lobe"] <- "Upper"
meta[results_list$lower_lobe_cells, "lobe"] <- "Lower"

## Run DE between tumor lobes on tumor cells only

# Pre-de find neighbors
pre_de_obj <- pre_de(
  metadata = meta,
  cell_type_metadata_colname = "major_cell_type",
  mm_radius = 0.05,
  sdimx_colname = "x_slide_mm",
  sdimy_colname = "y_slide_mm",
  verbose = TRUE,
  adjacencies_only = TRUE
)

# Add total counts to metadata as DE expects
meta$totalcounts = meta$nCount_RNA

# Use overlap metric to identify genes to test
overlap_metrics <- 
smiDE::overlap_ratio_metric(assay_matrix = seu.obj@assays$RNA$data,
                           metadata = meta,
                           cluster_col = "major_cell_type",
                           radius = 0.05,
                           #split_neighbors_by_colname = "Run_Tissue_name",
                           sdimx_col = "x_slide_mm",
                           sdimy_col = "y_slide_mm"
                           )

# Save overlap metrics
saveRDS(overlap_metrics, file = paste0(tumorDE_dir, "/Data/overlap_metrics.rds"))

# Only analyze genes with ratio less than or equal to 1
analysis_genes <- overlap_metrics[major_cell_type=="Tumor" & ratio <= 1]$target

# DE test dependence on tumor lobe
de_obj <- smi_de(
  assay_matrix = seu.obj@assays$RNA@counts[analysis_genes,],
  metadata = meta[which(meta[,"major_cell_type"] == "Tumor"),],
  formula = ~RankNorm(otherct_expr) + lobe + offset(log(totalcounts)),
  pre_de_obj = pre_de_obj,
  groupVar = "lobe",
  neighbor_expr_cell_type_metadata_colname = "major_cell_type",
  neighbor_expr_totalcount_normalize = TRUE,
  neighbor_expr_totalcount_scalefactor = NULL,
  family = "nbinom2",
  targets = analysis_genes,
  nCores = 20
)

# Save objects
saveRDS(pre_de_obj, file = paste0(tumorDE_dir, "/Data/pre_de_obj.rds"))
saveRDS(de_obj, file = paste0(tumorDE_dir, "/Data/de_obj_tumorGenes.rds"))

de_obj <- readRDS(paste0(tumorDE_dir, "/Data/de_obj_tumorGenes.rds"))

# Find gene sets

  filter(p.adj < 0.05) %>%
  filter(log2(fold_change) < -0.5) %>%
  select(target)

  filter(p.adj < 0.05) %>%
  filter(log2(fold_change) > 0.5) %>%
  select(target)

pathway_res <- list()

  print(paste0("Top pathways in set ", i, ":"))
  gene_ids <- bitr(get(i)$target, "SYMBOL", "ENTREZID", org.Hs.eg.db)
  enrich_result1 <- enrichPathway(gene = gene_ids$ENTREZID,
                                  organism = "human")
  pathway_res[i] <- enrich_result1
  print(enrich_result1@result[1:10, c("Description", "FoldEnrichment", "qvalue")])
}

# Read in pathway results
pathway_res <- readRDS(file = paste0(tumorDE_dir, "/Data/pathwayEnrichmentResults_tumorGenes.rds"))

# Merge top enriched pathways
enrich_paths <- rbind(pathway_res$upper_genes@result,
                      pathway_res$lower_genes@result)
enrich_paths$direction <- c(rep("upper", nrow(pathway_res$upper_genes@result)),
                            rep("lower", nrow(pathway_res$lower_genes@result)))


# Filter data frame for p.adjust < 0.1
df_filtered <- enrich_paths %>% filter(p.adjust < 0.1)

# Add a column for bar length based on direction
df_filtered <- df_filtered %>%
 mutate(bar_length = ifelse(direction == "upper", FoldEnrichment, -FoldEnrichment),
 color = ifelse(direction == "upper", -log10(p.adjust), -log10(p.adjust)))


# Arrange data frame by Fold Enrichment from highest to lowest
df_filtered <- df_filtered %>% arrange(desc(bar_length))


# Select top 5 in each direction by absolute bar length
df_top <- df_filtered %>%
  mutate(abs_length = abs(bar_length)) %>%
  group_by(direction) %>%
  slice_max(order_by = abs_length, n = 5, with_ties = FALSE) %>%
  ungroup()

# Convert lobe terminology
df_top$direction <- gsub("lower", "Dense", df_top$direction)
df_top$direction <- gsub("upper", "Interdigitated", df_top$direction)


# Custom shorten a few pathway names for plotting
df_top$Description <- dplyr::recode(df_top$Description,
                                    "Modulation of host responses by IFN-stimulated genes" = "*Modulation by IFN",
                                    "Deposition of new CENPA-containing nucleosomes at the centromere" = "*Nucleosome deposition",
                                    "alpha-linolenic (omega3) and linoleic (omega6) acid metabolism" = "*ALA (omega3) and LA (omega6) metabolism")

# Create the plot
ggplot(df_top, aes(x = reorder(Description, bar_length), y = bar_length, fill = direction)) +
 geom_bar(stat = "identity") +
 scale_fill_manual(values = c("Dense" = "lightskyblue3", "Interdigitated" = "tomato3")) +
 coord_flip() +
 theme_minimal() +
 labs(y = "Fold Enrichment", x = "Pathway", fill = "Lobe") +
 theme(axis.text.y = element_text(size = 12),
 axis.text.x = element_text(size = 12),
 axis.title.y = element_text(size = 14),
 axis.title.x = element_text(size = 14),
 legend.text = element_text(size = 14),
 legend.title = element_text(size = 20))
