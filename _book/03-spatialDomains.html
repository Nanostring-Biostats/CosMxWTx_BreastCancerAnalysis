3  Spatial domains

In this section, we use Novae to identify spatial domains within this dataset.

As this package is written in python, we start by generating an adata object from our filtered Seurat object.

Next, we switch into Python and run novae to generate spatial domains.

Show the code 
import novae
import igraph
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import scanpy as sc
import numpy as np
import os
import hdf5plugin

sc.settings.figdir = os.path.join(r.niche_dir, "Plots")
novae.settings.auto_preprocessing = True


# Read in sample
adata = sc.read_h5ad("analysis/Robjects/assay_data.h5ad")

## convert spatial from mm to microns for novae

# Extract coordinates from obs
x_mm = adata.obs["x_slide_mm"].to_numpy()
y_mm = adata.obs["y_slide_mm"].to_numpy()

# Create spatial_mm matrix
spatial_mm = np.column_stack((x_mm, y_mm))
adata.obsm["spatial_mm"] = spatial_mm

# Convert to microns
spatial_um = spatial_mm * 1000
adata.obsm["spatial_um"] = spatial_um

# Duplicate spatial_um to spatial
adata.obsm["spatial"] = spatial_um

## Start running novae
# Create a Delaunay graph of the data
novae.spatial_neighbors(adata, radius=80, coord_type='generic')

# Use a quick plot to ensure that neighbors were calculated appropriately
# Looking for very few red (unconnected) cells that are only on the edges of tissue
novae.plot.connectivities(adata)

# Load in pre-trained model
model1 = novae.Novae.from_pretrained("MICS-Lab/novae-human-0")

# Fine tune model based on CosMx data in this study
model1.fine_tune(adata, max_epochs=50, logger=True) 
model1.compute_representations(adata, zero_shot=False) 

for i in range(1, 20):
  model1.assign_domains(adata, i)

model1_dir = os.path.join(r.niche_dir, "novae_model_one")
model1.save_pretrained(save_directory=model1_dir)

import matplotlib.pyplot as plt
dh = model1.plot_domains_hierarchy()
plt.savefig(os.path.join(r.niche_dir, "Plots", "novae_model1_hierarchy.png"))
plt.close()


adata.write_h5ad(
  os.path.join(r.objects_dir, "anndata-plus-novae.h5ad"),
  compression=hdf5plugin.FILTERS["zstd"],
  compression_opts=hdf5plugin.Zstd(clevel=5).filter_options
)

Plot top pathways per domain, using HieraType marker heatmap code.

Show the code 
# Read in pathway scores
pwd <- readRDS(file = paste0(pathway_dir, "/Data/cellsAUC_full.rds"))

if (!("seu.obj" %in% ls())){
  seu.obj <- readRDS(results_list$seurat_file_postCellTyping)
}

# Read in named novae domains
novae_9 <- readRDS(paste0(niche_dir, "/Data/novae_domains_9_namedAssignments.rds"))

# Extract metadata and add novae domains
df <- seu.obj@meta.data
df$novae_domain_9 <- novae_9[df$cell_ID]
if(all(df$cell_ID) == colnames(pwd)){
  
  ## Heatmap of pathways by domain
  pwfctbl <- 
    HieraType::clusterwise_foldchange_metrics(normed = pwd
                                              ,metadata = df[,c("novae_domain_9", "cell_ID")]
                                              ,cluster_column = "novae_domain_9")
  
  
  # Save it
  saveRDS(pwfctbl, file = paste0(niche_dir, "/Data/pwfctbl_novae9.rds"))
  
  # Remove NA cluster and 'reactome' from pathway names
  pwfctbl <- pwfctbl[!(is.na(pwfctbl$cluster)),]
  pwfctbl$gene <- gsub("REACTOME-", "", pwfctbl$gene)
  
  # Replace some pathway names with shorter abbreviations for better plotting
  pwfctbl$gene <- dplyr::recode(pwfctbl$gene,
                                "TP53-REGULATES-TRANSCRIPTION-OF-SEVERAL-ADDITIONAL-CELL-DEATH-GENES-WHOSE-SPECIFIC-ROLES-IN-P53-DEPENDENT-APOPTOSIS-REMAIN-UNCERTAIN" = "(*)TP53-REGULATED-CELL-DEATH",
                                "CROSS-PRESENTATION-OF-PARTICULATE-EXOGENOUS-ANTIGENS-PHAGOSOMES" = "(*)EXOGENOUS-ANTIGENS-PRESENTATION",
                                "NR1H2-NR1H3-REGULATE-GENE-EXPRESSION-LINKED-TO-LIPOGENESIS" = "(*)NR1H2-NR1H3-REGULATED-LIPOGENESIS",
                                "DEFECTIVE-CHST14-CAUSES-EDS-MUSCULOCONTRACTURAL-TYPE" = "(*)DEFECTIVE-CHST14-CAUSES-EDS")
  
  # Change pathway names to first letter capitalized
  pwfctbl$gene <- stringr::str_to_sentence(pwfctbl$gene)
  pwfctbl$gene <- gsub("-", " ", pwfctbl$gene)
  
  ## Plot it as a tile plot since fraction of cells doesn't have much meaning for a pathway
  # Extract features to use and order as in HieraType code
  fctable <- data.table::copy(pwfctbl)
  fctable[, `:=`(scaled_fold_change, fold_change/max(fold_change)), 
          by = gene]
  fctable[, `:=`(scaled_expr, cluster_expr/max(cluster_expr)), 
          by = gene]
  fctable[is.na(scaled_fold_change), `:=`(scaled_fold_change, 
                                          0)]
  fctable[is.na(scaled_expr), `:=`(scaled_expr, 0)]
  dmat <- data.table::dcast(fctable, gene ~ cluster, 
                            value.var = "scaled_fold_change")
  fc_mat <- as.matrix(dmat[, -c("gene"), with = FALSE])
  rownames(fc_mat) <- dmat[, gene]
  
  featsuse <- unique(c(fctable[order(-fold_change)][cluster_prop > 
                                                      0.05, head(.SD, 3), by = cluster][, 
                                                                                        unique(gene)]))
  
  suppressMessages(hmap <- ComplexHeatmap::Heatmap(fc_mat[featsuse, 
  ]))
  suppressMessages(grid::grid.grabExpr(hmap <- ComplexHeatmap::draw(hmap)))
  
  gene_order <- featsuse[ComplexHeatmap::row_order(hmap)]
  
  ct_order <- colnames(fc_mat)[ComplexHeatmap::column_order(hmap)]
  
  pd <- fctable[gene %in% featsuse]
  pd[, `:=`(gene, factor(gene, levels = gene_order))]
  pd[, `:=`(cluster, factor(cluster, levels = ct_order))]
  
  
  pd[, `:=`(rnk, rank(-fold_change)), by = gene]
  pd[, `:=`(gene, factor(gene, levels = pd[rnk == 1][order(cluster, 
                                                           -fold_change)][, unique(gene)]))]
  
  
  ggplot(as.data.frame(pd),
         aes(y = cluster, x = gene, fill = scaled_expr)) +
    geom_tile() +
    scale_fill_gradientn(colors = RColorBrewer::brewer.pal(9, "Blues"), 
                         name = "Scaled Pathway Score") +
    theme_classic() +
    theme(axis.text.x = element_text(angle = 90, hjust = 1, vjust = 0.5, 
                                     size = 12),
          axis.title.x = element_blank(),
          axis.text.y = element_text(size = 12),
          axis.title.y = element_blank()) +
    coord_flip()
  ggsave(filename = paste0(niche_dir, 
                           "/Plots/heatmap_pathways_by_domain_formatted.png"),
         height = 5, width = 8, units = "in", dpi = 300)
  
  
}
