# Use a smoothing function modified from https://nanostring-biostats.github.io/CosMx-Analysis-Scratch-Space/posts/marker-gene-smoothing/
xy_nn <- function(xy, n_neighbors=50){
  xy <- as.data.table(xy)
 
    ## identify nearest n_neighbors (+1 includes the cell as a neighbor to itself)
  nn_xy <- RANN::nn2(xy[,.(x_slide_mm, y_slide_mm)],k = n_neighbors + 1)$nn.idx
  nn_xy <- data.table::melt(cbind(xy[,.(cell_ID)], data.table(nn_xy))
                              , id.vars=c("cell_ID", "V1"))
  colnames(nn_xy) <- c("cell_ID1", "cell_ID1_idx", "neighbor", "cell_ID2_idx")
  nn_xy <- merge(nn_xy
                   , nn_xy[,.(cell_ID2=cell_ID1, cell_ID2_idx=cell_ID1_idx)][
                     ,unique(.SD)],by="cell_ID2_idx")
  
  ## Cell x cell neighbor indicator matrix
  wxy <- Matrix::sparseMatrix(i = c(unique(nn_xy$cell_ID1_idx), nn_xy$cell_ID2_idx)
                                ,j=c(unique(nn_xy$cell_ID1_idx), nn_xy$cell_ID1_idx)
                                ,x=1)
  
  rownames(wxy) <- colnames(wxy) <- nn_xy[order(nn_xy$cell_ID1_idx),unique(cell_ID1)]
  
  ## Column standardize, so that columns (cells) sum to 1, and each neighbor given equal weight.
  mxy <- Matrix::sparseMatrix(i=1:ncol(wxy)
                                ,j=1:ncol(wxy)
                                ,x=1/Matrix::colSums(wxy)
                                )
  dimnames(mxy) <- dimnames(wxy)
  smoother <- mxy %*% wxy
  smoother <- smoother[,xy$cell_ID]
  return(smoother)
}

# Get smoother for tumor cells only

  pairwise.df$term = gsub("broad_niche_v2", "broad_niche", pairwise.df$term)

# Save pairwise results to a table to be included as supplementary information.
write.csv(as.data.frame(pairwise.df)[,c(setdiff(colnames(pairwise.df), c("df", "null", "fold_change")))],
          file = paste0(immNeigh_dir, "/Data/CD8TcytotoxicBroadNicheDE_table.csv"),

de_genes <- pairwise.df %>% filter(fdr < 0.05) %>% pull(target) %>% unique()

# Extract estimated means for all de genes

# Filter for significant genes (FDR < 0.05) and count number and type of DE comparisons per gene
de_signature <- pairwise.df %>%
  filter(fdr < 0.05) %>%
  distinct(target, contrast) %>%
  group_by(target) %>%
  summarise(
    de_signature = paste(sort(unique(contrast)), collapse = "_"),
