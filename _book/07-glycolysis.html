7  Glycolysis

In this section we focus on glycolysis and the impact of the surrounding glycolytic environment on gene expression.

7.1 Smoothed glycolysis pathway scores

We start by looking at glycolysis in space. We use pathway calculations from AUCell to generate glycolysis pathway scores for every cellular neighborhood.

# Use a smoothing function modified from https://nanostring-biostats.github.io/CosMx-Analysis-Scratch-Space/posts/marker-gene-smoothing/
xy_nn <- function(xy, n_neighbors=50){
  xy <- as.data.table(xy)
 
    ## identify nearest n_neighbors (+1 includes the cell as a neighbor to itself)
  nn_xy <- RANN::nn2(xy[,.(x_slide_mm, y_slide_mm)],k = n_neighbors + 1)$nn.idx
  nn_xy <- data.table::melt(cbind(xy[,.(cell_ID)], data.table(nn_xy))
                              , id.vars=c("cell_ID", "V1"))
  colnames(nn_xy) <- c("cell_ID1", "cell_ID1_idx", "neighbor", "cell_ID2_idx")
  nn_xy <- merge(nn_xy
                   , nn_xy[,.(cell_ID2=cell_ID1, cell_ID2_idx=cell_ID1_idx)][
                     ,unique(.SD)],by="cell_ID2_idx")
  
  ## Cell x cell neighbor indicator matrix
  wxy <- Matrix::sparseMatrix(i = c(unique(nn_xy$cell_ID1_idx), nn_xy$cell_ID2_idx)
                                ,j=c(unique(nn_xy$cell_ID1_idx), nn_xy$cell_ID1_idx)
                                ,x=1)
  
  rownames(wxy) <- colnames(wxy) <- nn_xy[order(nn_xy$cell_ID1_idx),unique(cell_ID1)]
  
  ## Column standardize, so that columns (cells) sum to 1, and each neighbor given equal weight.
  mxy <- Matrix::sparseMatrix(i=1:ncol(wxy)
                                ,j=1:ncol(wxy)
                                ,x=1/Matrix::colSums(wxy)
                                )
  dimnames(mxy) <- dimnames(wxy)
  smoother <- mxy %*% wxy
  smoother <- smoother[,xy$cell_ID]
  return(smoother)
}

# Smoothed glycolysis pathway

7.2 Differential expression as a function of neighborhood glycolysis

Here we run DE on every cell type to try to understand how every cell type changes as a function of glycolysis levels in the neighborhood it finds itself in. The prediction is that we might find some genes that are consistently up or down regulated across all or many cell types. This might also point us towards cell types that are more or less responsive / susceptible to the surrounding glycolysis environment. As additional factors in our model, we also include niche and the number of each tumor cell type in the 50 nearest neighbors. For speed in making this many comparisons, we work with normalized counts and a gaussian model.

meta$smoothed_glyc <- smoothed_glyc[1,meta$cell_id]

# Add number of each type of tumor cell in 50 nearest neighbors
coords <- as.matrix(meta[, c("x_slide_mm", "y_slide_mm")])
K <- 50
idx <- FNN::get.knn(coords, k = K)$nn.index  # (N x K), self is not included here

# Count cells in neighborhood
types   <- as.character(meta$minor_cell_type)
targets <- c("LUM+ tumor","HPD+ tumor","PGC+ tumor","KRT16+ tumor","Proliferating tumor")

neighbor_counts <- lapply(targets, function(lbl) {
  is_target <- (types == lbl)
  mat <- matrix(is_target[idx], nrow = nrow(idx), ncol = ncol(idx))
  rowSums(mat)
})
names(neighbor_counts) <- targets

meta$neighboring_lum  <- neighbor_counts[["LUM+ tumor"]]
meta$neighboring_hpd  <- neighbor_counts[["HPD+ tumor"]]
meta$neighboring_pgc  <- neighbor_counts[["PGC+ tumor"]]
meta$neighboring_krt  <- neighbor_counts[["KRT16+ tumor"]]

For every cell type, we make a volcano plot showing the estimated difference in expression between high and low glycolysis neighborhoods for each tested gene. Genes whose difference reached significance are colored and a subset are labeled. Also generate a table of DE results to be included as a supplementary table. Print each cell type to a separate sheet.

# Generate a list of pairwise DE results files
de_result_files <- dir(paste0(metabolism_dir, "/Data/"), pattern = "^ct_.*smoothedGlycolysis_de_obj_plusNicheNeighborTumor\\.rds$")

# Make a list of results to print to table
res_list <- list()

#de_res_list <- list()
for(de_file in de_result_files){
  de_obj <- readRDS(paste0(metabolism_dir, "/Data/", de_file))
  
  # Extract cell type
  ct <- gsub("ct_", "", gsub("_smoothedGlycolysis_de_obj_plusNicheNeighborTumor.rds", "", de_file))
  
  # Extract pairwise results, which correspond to difference between mean and 1SD of glycolysis score
  pairwise.df <- results(de_obj, "pairwise")$pairwise
  pairwise.df <- pairwise.df %>% filter(term == "smoothed_glyc") 
  
  # Adjust p-value
  pairwise.df$p.adj <- p.adjust(pairwise.df$p.value, method = "fdr") # Use fdr for padj
  res_list[[ct]] <- as.data.frame(pairwise.df)[,c("target", "estimate", "SE", "df", "t.ratio", "p.value",
                                                  "fold_change", "modelest_cpc_1", "modelest_cpc_2",
                                                  "term", "p.adj")]
  
  # Plot for each contrast, largely adapting smiDE volcano plot code
  for(contrast in unique(pairwise.df$contrast)){
    pairwise.df_i <- as.data.frame(pairwise.df)[which(pairwise.df$contrast == contrast),]

    # Extract comparison
    cc = pairwise.df_i$contrast[1]
    compv = compv <- strsplit(as.character(cc), "\\ vs.\\ ")[[1]]
    if(grepl(" [\\/-]\\ ", cc)) compv <- strsplit(as.character(cc), "\\ [\\/-]\\ ")[[1]]

    ## Plot with scaled estimates to adjust for expression levels

    # Large estimates disrupt scaling, so cap them and change the shape
    pairwise.df_i$scaled_estimate <- pairwise.df_i$estimate / pairwise.df_i$modelest_cpc_2
    pairwise.df_i$scaled_estimate_capped <- pmax(pmin(pairwise.df_i$scaled_estimate, 0.95), -0.95)
    pairwise.df_i$capped <- abs(pairwise.df_i$scaled_estimate) > 0.95

    # x labels
    xrng <- c((max(abs(pairwise.df_i$scaled_estimate_capped)) * -1),
              max(abs(pairwise.df_i$scaled_estimate_capped)))
    xrng <- pretty(xrng)

    # Set up y labels
    nDigits <- function(x) suppressWarnings(as.integer(strsplit(x = formatC(formatC(x = x, format="e"), format = "s"), split = "e")[[1]][2]))
    pairwise.df_i <- as.data.table(pairwise.df_i)
    bonf<- pairwise.df_i[,0.05/(uniqueN(target) * uniqueN(contrast))]
    max.y <- nDigits(min(pairwise.df_i[,min(p.value)+10^-300], signif(bonf / 2, 1)))

    ybrk <- 1 * 10^pretty( c(-1:(max.y)) )
    ylab <- ybrk

    # Add coloring
    est_cutoff = 0.1
    pairwise.df_i$sig_group = paste0("P adj > 0.05\nand/or\nScaled exp diff < ", est_cutoff)
    pairwise.df_i$sig_group[intersect(which(pairwise.df_i$p.adj < 0.05),
                                      which((pairwise.df_i$scaled_estimate) > est_cutoff))] <- compv[1]
    pairwise.df_i$sig_group[intersect(which(pairwise.df_i$p.adj < 0.05),
                                      which((pairwise.df_i$scaled_estimate) < -est_cutoff))] <- compv[2]
    my_cols = c("tomato3", "lightskyblue3", "lightgrey")
    names(my_cols) = c(compv[1],
                       compv[2],
                       paste0("P adj > 0.05\nand/or\nScaled exp diff < ", est_cutoff))

    # Set up yintercept for pval cutoff
    pval_cutoff_y <- pairwise.df %>% filter(p.adj < 0.05) %>% pull(p.value) %>% max()

    # Plot with top DE genes labeled
    ggplot(pairwise.df_i,
           aes(x = scaled_estimate_capped, y = p.value + 10^-300)) +
      geom_hline(yintercept = pval_cutoff_y, linetype = 2) +
      geom_vline(xintercept = est_cutoff, linetype = 2) +
      geom_vline(xintercept = -est_cutoff, linetype = 2) +
      geom_point(aes(color = sig_group, shape = capped)) +
      scale_shape_manual(values = c(`FALSE` = 19, `TRUE` = 17)) +
      scale_color_manual(values = my_cols, name = "Significance group") +
      scale_y_continuous(trans = smiDE:::change_axis_revlog_trans(base=10),
                         breaks = ybrk,
                         labels = formatC(ylab, format = "e", digits = 0)) +
      scale_x_continuous(breaks = xrng, limits = range(xrng)) +
      theme_bw() +
      ylab("p value") +
      xlab(paste0("Lower neighborhood glycolysis ", sprintf('\u2190'),
                  " Scaled estimated expression difference ",
                  sprintf('\u2192'), "Higher neighborhood glycolysis" )) +
      geom_text(data = pairwise.df[1,],
                aes(label = paste0(sum(pairwise.df_i$sig_group == compv[1]), "\ngenes"),
                    x = max(xrng)*0.8, y = sqrt(min(ybrk))),
                color = my_cols[[compv[1]]]) +
      geom_text(data = pairwise.df[1,],
                aes(label = paste0(sum(pairwise.df_i$sig_group == compv[2]), "\ngenes"),
                    x = min(xrng)*0.8, y = sqrt(min(ybrk))),
                color = my_cols[[compv[2]]]) +
      ggtitle(paste0(ct,
                     " gene expression as a function of glycolysis in the neighborhood")) +
      theme(plot.title = element_text(size = 8)) +
      theme(axis.title.x = element_text(size = 6)) +
      theme(axis.title.y = element_text(size = 6)) +
      geom_label_repel(data = pairwise.df_i %>% filter(!grepl("P adj", sig_group)),
                       aes(x = scaled_estimate_capped, y = p.value + 10^-300,
                           label = target)) +
      guides(shape = "none")
    ggsave(filename = paste0(metabolism_dir, "/Plots/", gsub("+", "", ct), "_volcano_GLYCneighborhoodDifferences_scaledEstDiff_", compv[1], "vs",
                             compv[2], "_withGeneLabels_v2.png"))
  }
}

de_result_files <- dir(paste0(metabolism_dir, "/Data/"), pattern = "^ct_.*smoothedGlycolysis_de_obj_plusNicheNeighborTumor\\.rds$")

de_res_list <- list()
for(de_file in de_result_files){
  res <- readRDS(paste0(metabolism_dir, "/Data/", de_file))

  pairwise.df <- pairwise.df %>% filter(term == "smoothed_glyc")

# Define a function to add the adjusted p-values
add_p_adj <- function(df) {
  df$p.adj <- p.adjust(df$p.value, method = "fdr") # Note the method used here
  return(df)
}

# Apply the function to each data frame in the list
de_res_list <- lapply(de_res_list, add_p_adj)


## Barplot showing the number of genes that are up or down regulated per cell type
# count significant genes
sig_genes <- list()

# Iterate through each data frame in the list
for (ct_name in names(de_res_list)) {
  df <- de_res_list[[ct_name]]
  
  # Filter for significant genes
  significant_genes <- df %>% filter(p.adj < 0.05) %>% filter(abs(estimate/modelest_cpc_2) > 0.1)
  
  # Count upregulated genes
  upregulated <- sum(significant_genes$estimate/significant_genes$modelest_cpc_2 > 0.1)
  
  # Count downregulated genes
  downregulated <- sum(significant_genes$estimate/significant_genes$modelest_cpc_2 < -0.1)
  
  # Count total genes evaluated
  tot_genes <- nrow(df)
  
  # Store the results in a data frame
  sig_genes[[ct_name]] <- data.frame(
    ct = ct_name,
    direction = c("up", "down"),
    nGenes = c(upregulated, downregulated),
    totGenes = tot_genes
  )
}
